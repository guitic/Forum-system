238:   function getRedirect() {
239:     var params = new URLSearchParams(window.location.search);
240:     var r = params.get("redirect");
241:     if (!r) return defaultRedirect;
242:     // 仅允许相对路径，避免开放重定向
243:     try {
244:       var u = new URL(r, window.location.origin);
245:       if (u.origin !== window.location.origin) return defaultRedirect;
246:       if (!/^(index|post|auth)\.html/.test(u.pathname.split("/").pop())) return defaultRedirect;
247:       return u.pathname + u.search;
248:     } catch (e) {
249:       return defaultRedirect;
250:     }
251:   }

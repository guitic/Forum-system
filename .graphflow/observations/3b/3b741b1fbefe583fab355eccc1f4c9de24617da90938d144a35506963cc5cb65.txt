152: def parse_log_ts(line: str) -> Optional[_dt.datetime]:
153:     """从 Nginx access log 中解析时间戳 (支持常见格式)。"""
154:     # Nginx 默认格式: 2024/01/01 12:34:56 +0800
155:     m = re.search(r"\[([^\]]+)\]", line)
156:     if not m:
157:         return None
158:     raw = m.group(1)
159:     for fmt in (
160:         "%d/%b/%Y:%H:%M:%S %z",
161:         "%Y-%m-%d %H:%M:%S",
162:         "%Y-%m-%dT%H:%M:%S",
163:     ):
164:         try:
165:             return _dt.datetime.strptime(raw, fmt)
166:         except ValueError:
167:             continue
168:     return None

171: def tail_lines(path: str, since: Optional[_dt.datetime] = None) -> List[str]:
172:     """读取日志，可按时间窗口过滤。"""
173:     p = Path(path)
174:     if not p.exists():
175:         return []
176:     try:
177:         with p.open("r", encoding="utf-8", errors="replace") as f:
178:             lines = f.readlines()
179:     except Exception as exc:
180:         LOG.warning("读取日志失败 %s: %s", path, exc)
181:         return []
182:     if since is None:
183:         return lines
184:     result: List[str] = []
185:     for ln in lines:
186:         ts = parse_log_ts(ln)
187:         if ts is None:
188:             # 无时间戳的行（如 error.log 多行），保守保留
189:             result.append(ln)
190:             continue
191:         if ts >= since:
192:             result.append(ln)
193:     return result

196: def parse_access_log_status(line: str) -> Optional[int]:
197:     """解析 access log 中的 HTTP 状态码。"""
198:     # 标准 Nginx 格式: ... "GET / HTTP/1.1" 200 ...
199:     m = re.search(r'"[^"]*"\s+(\d{3})\s', line)
200:     if m:
201:         try:
202:             return int(m.group(1))
203:         except ValueError:
204:             return None
205:     # 兼容自定义格式: 状态码作为单独的字段
206:     m = re.search(r"\s(\d{3})\s+(\d+)", line)
207:     if m:
208:         try:
209:             return int(m.group(1))
210:         except ValueError:
211:             return None
212:     return None

347: def check_disk() -> CheckResult:
348:     """检查磁盘使用率。"""
349:     code, out, _ = run_cmd("df -P /opt / 2>/dev/null")
350:     threshold = CONFIG["disk_threshold"]
351:     worst = 0
352:     worst_path = ""

366:         name="磁盘使用率",

374:     """检查内存使用率。"""

386:         name="内存使用率",

393: def check_load() -> CheckResult:
394:     """检查 CPU 负载。"""
395:     code, out, _ = run_cmd("cat /proc/loadavg 2>/dev/null")
396:     try:
397:         load1, load5, load15 = (float(x) for x in out.split()[:3])

425: class AlertState:
426:     """记录每个检查项的连续失败次数，用于抑制重复告警。"""
427: 
428:     def __init__(self, state_dir: str) -> None:
429:         self.dir = Path(state_dir)
430:         self.dir.mkdir(parents=True, exist_ok=True)
431:         self.path = self.dir / "alert_state.json"
432:         self.data: Dict[str, int] = {}
433:         if self.path.exists():
434:             try:
435:                 self.data = json.loads(self.path.read_text(encoding="utf-8"))
436:             except Exception:
437:                 self.data = {}
438: 
439:     def increment(self, name: str) -> int:
440:         self.data[name] = self.data.get(name, 0) + 1
441:         self._save()
442:         return self.data[name]
443: 
444:     def reset(self, name: str) -> None:
445:         self.data.pop(name, None)
446:         self._save()
447: 
448:     def get(self, name: str) -> int:
449:         return self.data.get(name, 0)
450: 
451:     def _save(self) -> None:
452:         try:
453:             self.path.write_text(
454:                 json.dumps(self.data, ensure_ascii=False, indent=2),
455:                 encoding="utf-8",
456:             )
457:         except Exception as exc:
458:             LOG.warning("保存告警状态失败: %s", exc)

500: def render_report(results: List[CheckResult],
501:                   consecutive: Optional[Dict[str, int]] = None,
502:                   ) -> Tuple[str, str]:
503:     """返回 (html, text) 双格式报告。"""
504:     now = _dt.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
505:     host = os.uname().nodename
506:     critical = [r for r in results if r.alert]
507:     status = "🟢 正常" if not critical else "🔴 异常"
508: 
509:     # 文本格式
510:     text_lines = [
511:         f"论坛系统巡检报告  {now}",
512:         f"主机: {host}    总体状态: {status}",
513:         "=" * 60,
514:     ]
515:     for r in results:
516:         flag = "✅" if r.ok else ("🚨" if r.alert else "⚠️")
517:         text_lines.append(f"{flag} [{r.name}]")
518:         for dline in r.detail.splitlines():
519:             text_lines.append(f"    {dline}")
520:         if consecutive:
521:             cnt = consecutive.get(r.name, 0)
522:             if cnt:
523:                 text_lines.append(f"    连续失败: {cnt} 次")
524:         text_lines.append("")
525: 
526:     # HTML 格式
527:     html_rows = []
528:     for r in results:
529:         color = "#28a745" if r.ok else ("#dc3545" if r.alert else "#ffc107")
530:         icon = "✅" if r.ok else ("🚨" if r.alert else "⚠️")
531:         cnt = (consecutive or {}).get(r.name, 0)
532:         cnt_html = f'<div style="font-size:0.85em;color:#666;">连续失败 {cnt} 次</div>' if cnt else ""
533:         html_rows.append(
534:             f'<tr>'
535:             f'<td style="padding:10px;border-bottom:1px solid #eee;">{icon} {r.name}</td>'
536:             f'<td style="padding:10px;border-bottom:1px solid #eee;">'
537:             f'<span style="color:{color};font-weight:bold;">'
538:             f'{"正常" if r.ok else "告警"}</span>'
539:             f'<br><small style="color:#666;">{r.detail.replace(chr(10), "<br>")}</small>'
540:             f'{cnt_html}</td>'
541:             f'</tr>'
542:         )
543: 
544:     html = f"""<!DOCTYPE html>
545: <html><head><meta charset="utf-8">
546: <style>
547: body {{ font-family: -apple-system, "Segoe UI", "Microsoft YaHei", sans-serif; max-width: 820px; margin: 24px auto; color: #333; }}
548: h1 {{ color: #333; border-bottom: 3px solid {'#dc3545' if critical else '#28a745'}; padding-bottom: 8px; }}
549: .meta {{ color: #666; margin-bottom: 16px; }}
550: table {{ width: 100%; border-collapse: collapse; }}
551: th {{ text-align: left; padding: 12px; background: #f8f9fa; }}
552: .footer {{ margin-top: 24px; color: #999; font-size: 0.85em; }}
553: </style></head><body>
554: <h1>论坛系统巡检报告</h1>
555: <div class="meta">时间: {now}　|　主机: {host}　|　总体状态: <b style="color:{'#dc3545' if critical else '#28a745'}">{status}</b></div>
556: <table><tr><th>检查项</th><th>结果</th></tr>{''.join(html_rows)}</table>

249:   function buildPageButtons(current, total) {
250:     // 显示 current 附近 5 个页码
251:     var buttons = [];
252:     var range = 2; // 当前页前后各 2 个
253:     var start = Math.max(1, current - range);
254:     var end = Math.min(total, current + range);
255: 
256:     if (start > 1) {
257:       buttons.push(1);
258:       if (start > 2) buttons.push("...");
259:     }
260:     for (var i = start; i <= end; i++) buttons.push(i);
261:     if (end < total) {
262:       if (end < total - 1) buttons.push("...");
263:       buttons.push(total);
264:     }
265:     return buttons;
266:   }

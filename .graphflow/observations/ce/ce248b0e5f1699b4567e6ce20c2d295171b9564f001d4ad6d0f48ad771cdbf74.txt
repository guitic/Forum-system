30: def require_auth(fn):
31:     """
32:     鉴权装饰器：从 Authorization: Bearer <token> 中解析 JWT，
33:     并将当前用户注入被装饰函数的 `current_user` 参数。
34:     """
35: 
36:     @functools.wraps(fn)
37:     def wrapper(*args, **kwargs):
38:         header = request.headers.get("Authorization", "")
39:         if not header.startswith("Bearer "):
40:             return jsonify({"error": "缺少有效的 Bearer Token"}), 401
41: 
42:         token = header[len("Bearer "):].strip()
43:         if not token:
44:             return jsonify({"error": "缺少有效的 Bearer Token"}), 401
45: 
46:         try:
47:             payload = jwt.decode(
48:                 token,
49:                 config.JWT_SECRET_KEY,
50:                 algorithms=[config.JWT_ALGORITHM],
51:             )
52:         except jwt.ExpiredSignatureError:
53:             return jsonify({"error": "Token 已过期，请重新登录"}), 401
54:         except jwt.InvalidTokenError:
55:             return jsonify({"error": "Token 无效"}), 401
56: 
57:         # 优先按 id 查用户；查询失败则视为无效
58:         try:
59:             user_id = int(payload.get("sub"))
60:         except (TypeError, ValueError):
61:             return jsonify({"error": "Token 无效"}), 401
62: 
63:         user = User.query.get(user_id)
64:         if not user:
65:             return jsonify({"error": "Token 对应的用户不存在"}), 401
66: 
67:         kwargs["current_user"] = user
68:         return fn(*args, **kwargs)
69: 
70:     return wrapper

106: def _reply_chain_info(reply, index):
107:     """一次回溯同时计算真实链信息。
108: 
109:     返回 (真实深度, 话题根回复 id)：
110:     - 真实深度：沿 parent_id 链向上回溯到一级回复的步数（一级回复 = 0）
111:     - 话题根 id：回溯到的那条一级回复的 id
112: 
113:     含环保护与悬空引用保护，避免脏数据导致死循环。
114:     """
115:     depth = 0
116:     cur = reply
117:     seen = set()
118:     while cur is not None and cur.parent_id:
119:         if cur.id in seen:
120:             break
121:         seen.add(cur.id)
122:         parent = index.get(cur.parent_id)
123:         if parent is None:
124:             break
125:         cur = parent
126:         depth += 1
127:     return depth, cur.id

130: def _prepare_reply_structure(replies, max_depth):
131:     """预计算回复树的展示结构。
132: 
133:     replies 必须已按 created_at 升序（父回复必先于子回复出现，便于迭代求解）。
134: 
135:     返回 (index, depth_map, root_map, children_map, display_parent)：
136:     - index:          id → Reply ORM 对象
137:     - depth_map:      id → 展示深度（已按 max_depth 收敛，0/1/2…）
138:     - root_map:       id → 话题根回复 id（真实值，一级回复为 None）
139:     - children_map:   展示父 id → [展示子 id, ...]
140:     - display_parent: id → 展示父 id（None 表示展示为一级）
141:     """
142:     index = {r.id: r for r in replies}
143: 
144:     # 1) 真实链信息，用于话题根 id 与溢出判定
145:     chain = {r.id: _reply_chain_info(r, index) for r in replies}
146: 
147:     # 2) 决定每个节点的"展示父节点"
148:     #    - 未超过深度上限：挂到真实父节点下
149:     #    - 超过深度上限：上提到父节点所在的展示层级，作为同级
150:     display_parent = {}
151:     for r in replies:
152:         if r.parent_id is None:
153:             display_parent[r.id] = None
154:             continue
155:         real_parent = index.get(r.parent_id)
156:         if real_parent is None:
157:             # 悬空引用（父回复已被单独删除等）：降级为一级回复
158:             display_parent[r.id] = None
159:             continue
160:         if chain[r.parent_id][0] + 1 <= max_depth:
161:             display_parent[r.id] = r.parent_id
162:         else:
163:             display_parent[r.id] = display_parent.get(r.parent_id)
164: 
165:     # 3) 展示深度（迭代求解，父节点必先于子节点）
166:     depth_map = {}
167:     for r in replies:
168:         parent_id = display_parent[r.id]
169:         depth_map[r.id] = 0 if parent_id is None else depth_map[parent_id] + 1
170: 
171:     # 4) 话题根 id（取真实值，一级回复为 None）
172:     root_map = {rid: info[1] for rid, info in chain.items()}
173: 
174:     # 5) 展示父子映射
175:     children_map = {}
176:     for r in replies:
177:         parent_id = display_parent[r.id]
178:         if parent_id is not None:
179:             children_map.setdefault(parent_id, []).append(r.id)
180: 
181:     return index, depth_map, root_map, children_map, display_parent

212: def _build_reply_tree(replies, post_author_id, max_depth):
213:     """构建展示用嵌套回复树。
214: 
215:     返回 (tree, flat)：
216:     - tree: 嵌套结构，每个节点含 children（最多 max_depth+1 层）
217:     - flat: 扁平列表（按时间升序），字段同 tree 节点，用于兼容与统计
218:     """
219:     index, depth_map, root_map, children_map, display_parent = _prepare_reply_structure(
220:         replies, max_depth
221:     )
222: 
223:     nodes = {
224:         r.id: _reply_node_dict(
225:             r, index, depth_map, root_map, children_map, post_author_id
226:         )
227:         for r in replies
228:     }
229:     flat = [nodes[r.id] for r in replies]
230: 
231:     def attach_children(node):
232:         node["children"] = [nodes[kid] for kid in children_map.get(node["id"], [])]
233:         for child in node["children"]:
234:             attach_children(child)
235:         return node
236: 
237:     tree = [
238:         attach_children(nodes[r.id])
239:         for r in replies
240:         if display_parent[r.id] is None
241:     ]
242:     return tree, flat

145:     // 携带 Bearer Token
146:     if (needsAuth && token) {
147:       headers["Authorization"] = "Bearer " + token;
148:     }
149: 
150:     // JSON body
151:     var body = undefined;
152:     if (opts.body !== undefined && opts.body !== null) {
153:       headers["Content-Type"] = "application/json";
154:       body = JSON.stringify(opts.body);
155:     }
156: 
157:     var url = buildUrl(path, opts.params);
158: 
159:     var response;
160:     var controller = new AbortController();
161:     var timeout = setTimeout(function () {
162:       controller.abort();
163:     }, 15000); // 15 秒超时
164: 
165:     try {
166:       response = await fetch(url, {
167:         method: method,
168:         headers: headers,
169:         body: body,
170:         credentials: "same-origin",
171:         signal: controller.signal,
172:       });
173:     } catch (netErr) {
174:       clearTimeout(timeout);
175:       if (netErr.name === "AbortError") {

446:   /**
447:    * 校验 URL 是否安全（阻止 javascript: / data: 等危险协议）
448:    */
449:   function isSafeUrl(url) {
450:     if (!url) return false;
451:     var u = url.trim().toLowerCase();
452:     // 阻止控制字符
453:     if (/[\u0000-\u001f\u007f]/.test(u)) return false;
454:     // 阻止 javascript: data: vbscript: 等
455:     if (/^(javascript|data|vbscript|file):/i.test(u)) return false;
456:     // 允许相对路径与锚点
457:     if (u.charAt(0) === "/" || u.charAt(0) === "." || u.charAt(0) === "#" || u.charAt(0) === "?") {
458:       return true;
459:     }
460:     // 必须以受信任协议开头
461:     for (var i = 0; i < SAFE_PROTOCOLS.length; i++) {
462:       if (u.indexOf(SAFE_PROTOCOLS[i]) === 0) return true;
463:     }
464:     return false;
465:   }

467:   /**
468:    * 解析属性字符串，过滤危险属性
469:    * @param {string} attrString
470:    * @param {string} tagName
471:    * @returns {string}
472:    */
473:   function filterAttrs(attrString, tagName) {
474:     if (!attrString) return "";
475:     var allowed = ALLOWED_ATTRS[tagName] || [];
476:     var result = [];
477:     var regex = /([a-zA-Z_:][\w:.-]*)\s*(?:=\s*("([^"]*)"|'([^']*)'|[^\s"'>]+))?/g;
478:     var m;
479:     while ((m = regex.exec(attrString)) !== null) {
480:       var name = m[1].toLowerCase();
481:       var value = m[3] !== undefined
482:         ? m[3]
483:         : m[4] !== undefined
484:           ? m[4]
485:           : m[5] !== undefined
486:             ? m[5]
487:             : "";
488: 
489:       // 跳过危险属性
490:       if (name.indexOf("on") === 0) continue;
491:       if (name === "style") continue;
492:       if (name === "formaction" || name === "srcdoc" || name === "xlink:href") continue;
493:       if (allowed.indexOf(name) === -1) continue;

495:       // 对 URL 属性做协议校验
496:       if (name === "href" || name === "src") {

498:       }

500:       result.push(name + '="' + escapeAttr(value) + '"');
501:     }
502:     return result.join(" ");
503:   }

533:   function sanitizeInner(html) {
534:     // 移除隐藏 HTML 注释（防止条件注释攻击等）
535:     html = html.replace(/<!--[\s\S]*?-->/g, "");
536:     // 移除 <?...?> 处理指令
537:     html = html.replace(/<\?[\s\S]*?\?>/g, "");
538: 
539:     var doc = new DOMParser().parseFromString("<body>" + html + "</body>", "text/html");
540:     var body = doc.body;
541: 
542:     // 先收集所有元素节点（避免遍历中修改树导致迭代状态错乱）。
543:     // 注意：body 属于 DOMParser 解析出的独立 document，
544:     // 不能用主文档的 Range.createNodeIterator 跨文档遍历，
545:     // 这里改用纯 DOM 递归遍历，兼容性最好。
546:     var allElements = [];
547:     (function collect(node) {
548:       var child = node.firstChild;
549:       while (child) {
550:         if (child.nodeType === 1) {
551:           allElements.push(child);
552:           collect(child);
553:         }
554:         child = child.nextSibling;
555:       }
556:     })(body);
557: 
558:     var DANGEROUS_TAGS = ["script", "style", "iframe", "object", "embed", "link", "meta", "base", "form", "input", "button", "select", "textarea"];
559: 
560:     allElements.forEach(function (node) {
561:       // 节点可能已被父节点替换/移除，跳过
562:       if (!node.parentNode) return;
563: 
564:       var tag = node.tagName.toLowerCase();
565:       if (!ALLOWED_TAGS[tag]) {
566:         if (DANGEROUS_TAGS.indexOf(tag) !== -1) {
567:           // 危险标签：整体删除（含子节点）
568:           node.parentNode.removeChild(node);
569:         } else {
570:           // 非危险标签：解包，保留文本内容
571:           // 用 body 所属文档创建片段，避免跨文档 Fragment 兼容性问题
572:           var frag = body.ownerDocument.createDocumentFragment();
573:           var child = node.firstChild;
574:           while (child) {
575:             frag.appendChild(child);
576:             child = node.firstChild;
577:           }
578:           node.parentNode.replaceChild(frag, node);
579:         }
580:         return;
581:       }
582: 
583:       // 对白名单标签做属性过滤
584:       var tagLower = tag;
585:       var allowed = ALLOWED_ATTRS[tagLower] || [];
586:       // 先读取属性，再逐个移除/重建
587:       var existing = [];
588:       for (var i = 0; i < node.attributes.length; i++) {
589:         var attrName = node.attributes[i].name;
590:         var attrVal = node.attributes[i].value;

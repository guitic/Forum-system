# ============================================================================
# 论坛系统 Crontab 配置示例
# ============================================================================
# 部署方式 (任选其一):
#   方案 A (推荐): 直接安装到 /etc/cron.d/ 目录，作为系统级定时任务
#       cp crontab.txt /etc/cron.d/forum
#       chmod 644 /etc/cron.d/forum
#   方案 B: 复制到 /etc/crontab 末尾
#       cat crontab.txt >> /etc/crontab
#   方案 C: 导入到 root 用户的 crontab
#       crontab -u root crontab.txt
#   方案 D: 导入到 forum 用户的 crontab
#       crontab -u forum crontab.txt
# ============================================================================
# 说明:
#   - 所有脚本默认路径: /opt/forum/scripts/
#   - 所有脚本日志默认路径: /var/log/forum/
#   - /etc/cron.d/ 下的任务必须带用户名 (root)
#   - crontab -u 安装的任务不需要用户名，但建议显式写 root 以增强可读性
#   - 邮件重定向 (>) 会覆盖默认邮件发送，可删除以启用系统默认邮件
# ============================================================================
# Crontab 字段说明 (以分钟数为单位):
#   ┌─ 分钟 (0-59)
#   │ ┌─ 小时 (0-23)
#   │ │ ┌─ 日 (1-31)
#   │ │ │ ┌─ 月 (1-12 或 JAN-DEC)
#   │ │ │ │ ┌─ 星期 (0-6 或 SUN-SAT, 0=SUN)
#   │ │ │ │ │
#   * * * * * 命令
# ============================================================================

# 环境变量 (适用于 /etc/cron.d/ 和 root crontab)
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
FORUM_APP_ROOT=/opt/forum
FORUM_REPO_DIR=/opt/forum/repo
FORUM_VENV_DIR=/opt/forum/venv
FORUM_LOG_DIR=/var/log/forum

# ============================================================================
# 1. 监控脚本 - 每 5 分钟执行一次
# ============================================================================
# 巡检 Nginx 日志、systemd 服务状态、系统资源，异常时通过 SMTP 告警
0,5,10,15,20,25,30,35,40,45,50,55 * * * * root /opt/forum/scripts/monitor.py >> /var/log/forum/monitor.cron.log 2>&1

# ============================================================================
# 2. Git 更新脚本 - 每日凌晨 3:00 自动执行
# ============================================================================
# 拉取最新代码、安装依赖、执行迁移、重启服务
# 建议配合 --clean 清理临时文件，配合 --dry-run 在测试阶段验证
0 3 * * * root /opt/forum/scripts/git_pull.sh --clean >> /var/log/forum/git_pull.cron.log 2>&1

# ============================================================================
# 3. 日志轮转 - 每日凌晨 4:00 清理 30 天前日志
# ============================================================================
0 4 * * * root find /var/log/forum -type f -name "*.log" -mtime +30 -delete 2>> /var/log/forum/cleanup.cron.log

# ============================================================================
# 4. 系统状态快照 - 每 15 分钟记录资源使用情况
# ============================================================================
*/15 * * * * root /bin/bash -c 'free -m > /var/log/forum/stats-$(date +\%F-\%H\%M).txt 2>&1'

# ============================================================================
# 5. Nginx 日志归档 - 每日 03:30 压缩前一天的日志
# ============================================================================
30 3 * * * root /bin/bash -c 'cd /var/log/nginx && tar czf forum_logs_$(date -d yesterday +\%F).tar.gz forum_access.log forum_error.log 2>/dev/null; cd - >/dev/null; mv /var/log/nginx/forum_logs_*.tar.gz /var/log/forum/archive/ 2>/dev/null'

# ============================================================================
# 6. 健康检查 - 每分钟验证后端 HTTP 探活 (可选)
# ============================================================================
# * * * * * root curl -sf -o /dev/null http://127.0.0.1:5000/healthz || /opt/forum/scripts/monitor.py --verbose >> /var/log/forum/healthcheck.log 2>&1

# ============================================================================
# 备注
# ============================================================================
# - 修改 crontab 后立即生效，无需重启 cron 服务
# - 使用 `crontab -e` (root) 或 `systemctl edit cron` 可交互编辑
# - 使用 `systemctl status cron` 确认 cron 服务运行
# - Rocky Linux 默认使用 cronie，配置目录为 /etc/cron.d/
# - 测试 crontab 是否生效: tail -f /var/log/cron
# ============================================================================
